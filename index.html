📋 Langkah Mudah Pasang Script di Google Sheets (1 Menit)
  1. Buka Google Drive > Buat spreadsheet baru di Google Sheets.
  2. Klik menu Extensions (Ekstensi) > Apps Script.
  3. Hapus semua kode yang ada di sana, lalu tempel (paste) kode berikut:
Code.gs 
/**
 * GOOGLE APPS SCRIPT - SISTEM REKAP INSPEKSI APAR & FSS
 * Pasang di: Google Sheets > Extensions > Apps Script
 * Lalu Deploy: Deploy > New deployment > Select type: Web App
 * - Execute as: Me (email Anda)
 * - Who has access: Anyone
 */

function doPost(e) {
  try {
    var rawData = e.postData.contents;
    var payload = JSON.parse(rawData);
    
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Inspeksi APAR");
    if (!sheet) {
      sheet = ss.insertSheet("Inspeksi APAR");
    }
    
    // Inisialisasi Header jika masih kosong
    if (sheet.getLastRow() === 0) {
      var headers = [
        "Waktu Input",
        "Tanggal Pemeriksaan",
        "Nama Pemeriksa",
        "Kode APAR",
        "Lokasi / Ruangan",
        "Gedung",
        "Lantai",
        "Merk",
        "Kapasitas",
        "Jenis Gas",
        "Kebersihan Tabung",
        "Indikator Tekanan",
        "Kunci Pengaman / Segel",
        "Selang Semprot",
        "Nozzle",
        "Tag / Label",
        "Status Akhir",
        "Catatan / Keterangan",
        "Foto Bukti Eviden (Drive)"
      ];
      sheet.appendRow(headers);
      
      var headerRange = sheet.getRange(1, 1, 1, headers.length);
      headerRange.setBackground("#1e293b");
      headerRange.setFontColor("#ffffff");
      headerRange.setFontWeight("bold");
      headerRange.setHorizontalAlignment("center");
      sheet.setFrozenRows(1);
    }
    
    // Buat folder foto di Google Drive
    var folderName = "Foto_Inspeksi_APAR";
    var folders = DriveApp.getFoldersByName(folderName);
    var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);
    
    var items = Array.isArray(payload.items) ? payload.items : [payload];
    var insertedCount = 0;
    
    items.forEach(function(item) {
      var photoUrl = "-";
      
      // Simpan foto Base64 ke Google Drive
      if (item.foto && item.foto.indexOf("data:image") === 0) {
        try {
          var parts = item.foto.split(",");
          var contentType = parts[0].split(":")[1].split(";")[0];
          var base64Data = parts[1];
          var decoded = Utilities.base64Decode(base64Data);
          var fileName = "APAR_" + (item.kode || "UNKNOWN") + "_" + Utilities.formatDate(new Date(), "Asia/Jakarta", "yyyyMMdd_HHmmss") + ".jpg";
          var blob = Utilities.newBlob(decoded, contentType, fileName);
          var file = folder.createFile(blob);
          file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
          photoUrl = file.getUrl();
        } catch (imgErr) {
          photoUrl = "Error foto: " + imgErr.toString();
        }
      } else if (item.foto && item.foto.indexOf("http") === 0) {
        photoUrl = item.foto;
      }
      
      var rowData = [
        new Date(),
        item.tgl || Utilities.formatDate(new Date(), "Asia/Jakarta", "dd/MM/yyyy HH:mm"),
        item.pemeriksa || "-",
        item.kode || item.id || "-",
        item.ruangan || "-",
        item.gedung || "-",
        item.lantai || "-",
        item.merk || "-",
        item.kapasitas || "-",
        item.jenisGas || "-",
        item.kebersihanTabung === "good" ? "Bersih" : (item.kebersihanTabung === "bad" ? "Kotor" : "-"),
        item.indikatorTekanan === "good" ? "Hijau (Normal)" : (item.indikatorTekanan === "bad" ? "Merah (Abnormal)" : "-"),
        item.kunciPengaman === "good" ? "Ada / Tersegel" : (item.kunciPengaman === "bad" ? "Tidak Ada / Rusak" : "-"),
        item.selangSemprot === "good" ? "Baik / Tidak Pecah" : (item.selangSemprot === "bad" ? "Pecah / Rusak" : "-"),
        item.nozzle === "good" ? "Ada / Bagus" : (item.nozzle === "bad" ? "Tidak Ada / Rusak" : "-"),
        item.tagLabel === "good" ? "Ada / Jelas" : (item.tagLabel === "bad" ? "Tidak Ada / Pudar" : "-"),
        item.status || "OK",
        item.keterangan || "-",
        photoUrl
      ];
      
      sheet.appendRow(rowData);
      insertedCount++;
    });
    
    return ContentService.createTextOutput(JSON.stringify({
      success: true,
      message: "Berhasil menyimpan " + insertedCount + " data ke Google Sheets & Drive.",
      folderUrl: folder.getUrl(),
      spreadsheetUrl: ss.getUrl(),
      insertedCount: insertedCount
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      success: false,
      error: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({
    status: "active",
    message: "Webhook Google Apps Script APAR siap menerima data."
  })).setMimeType(ContentService.MimeType.JSON);
}
