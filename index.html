☁️ Sinkronisasi Firebase Firestore

⚡
Database Akun Terhubung
Semua akun tersimpan di database server & dapat disinkronkan ke Firebase Firestore.
📖 Tutorial Wajib Pengaturan Firebase Rules Klik Buka/Tutup

Langkah Setting Rules di Firebase Console (Hanya 1x):

  1. Buka console.firebase.google.com > Pilih project apar-fss.
  2. Di menu kiri: Build > Firestore Database. (Jika belum ada, klik Create database dulu).
  3. Buka tab Rules (Aturan) di bagian atas.
  4. Hapus aturan lama, lalu tempel (*paste*) aturan berikut:
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /apar_log/{document=**} {
      allow read, write: if true;
    }
    match /users/{document=**} {
      allow read, write: if true;
    }
    match /{document=**} {
      allow read, write: if true;
    }
  }
}
⚠️ Setelah di-paste, klik tombol Publish (Publikasikan) di kanan atas. Tunggu 30 detik lalu coba sinkronkan kembali.

📊 Hubungkan Google Sheets & Drive

Hubungkan aplikasi APAR ke Google Sheets & Google Drive agar laporan inspeksi dan foto eviden otomatis tersimpan ke Cloud Google milik Anda.

Pastikan berakhiran /exec (bukan /edit).
📜 Panduan & Kode Script Google Sheets + Drive Klik Buka/Tutup

Langkah-Langkah Pembuatan di Google Sheets:

  1. Buka Google Sheets baru di browser.
  2. Di menu atas, pilih Extensions (Ekstensi) > Apps Script.
  3. Hapus semua kode lama, lalu paste (tempel) kode di bawah ini:
function toSafeCell(val) {
  if (val === null || val === undefined || val === '') return "-";
  if (typeof val === "object") return JSON.stringify(val);
  return String(val);
}

function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({
    success: true,
    message: "Google Apps Script Web App APAR K3 aktif & siap menerima data! ✅",
    timestamp: new Date().toISOString()
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var data = {};
    if (e && e.postData && e.postData.contents) {
      try {
        data = JSON.parse(e.postData.contents);
      } catch (jsonErr) {
        try {
          data = JSON.parse(decodeURIComponent(e.postData.contents));
        } catch (e2) {
          data = e.parameter || {};
        }
      }
    } else if (e && e.parameter) {
      data = e.parameter;
    }

    if (data && data.payload) {
      if (typeof data.payload === "string") {
        try { data = JSON.parse(data.payload); } catch(e3) {}
      } else if (typeof data.payload === "object") {
        data = data.payload;
      }
    }

    var ss = null;
    try {
      ss = SpreadsheetApp.getActiveSpreadsheet();
    } catch (eSs) {}

    if (!ss && data.spreadsheetId) {
      try {
        ss = SpreadsheetApp.openById(data.spreadsheetId);
      } catch (eId) {}
    }

    if (!ss) {
      return ContentService.createTextOutput(JSON.stringify({
        success: false,
        error: "Spreadsheet tidak terdeteksi. Pastikan Apps Script dibuat dari menu Extensions > Apps Script di dalam Google Sheets Anda!"
      })).setMimeType(ContentService.MimeType.JSON);
    }

    var sheet = ss.getSheetByName("Riwayat_APAR") || ss.getSheets()[0];
    if (sheet.getName() === "Sheet1" || sheet.getName() === "Lembar1") {
      try { sheet.setName("Riwayat_APAR"); } catch(eName) {}
    }

    // Inisialisasi Header Kolom jika sheet masih kosong
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Waktu", "Kode APAR", "Gedung", "Ruangan", "Lantai", 
        "Merk", "Kapasitas", "Gas", "Kebersihan", "Tekanan", 
        "Pin", "Selang", "Nozzle", "Tag", "Status", 
        "Keterangan", "Link Foto Drive", "Pemeriksa"
      ]);
      sheet.getRange(1, 1, 1, 18).setFontWeight("bold").setBackground("#0f172a").setFontColor("#ffffff");
    }

    // A. FITUR KIRIM MASSAL / SINKRONISASI MASSAL (sync_all)
    if (data.action === "sync_all" || Array.isArray(data.items)) {
      var items = data.items || [];
      if (!Array.isArray(items) && typeof items === "object") {
        var temp = [];
        for (var k in items) temp.push(items[k]);
        items = temp;
      }

      // Kosongkan baris data lama (mulai baris ke-2), tetap pertahankan Header
      var lastRow = sheet.getLastRow();
      if (lastRow > 1) {
        sheet.getRange(2, 1, lastRow - 1, 18).clearContent();
      }

      var rowsToAdd = [];
      for (var i = 0; i < items.length; i++) {
        var item = items[i];
        if (!item) continue;

        var photoUrl = toSafeCell(item.photoUrl || item.fotoUrl);
        if (item.foto && typeof item.foto === "string" && item.foto.indexOf("base64,") !== -1) {
          try {
            var folderName = "Foto_Eviden_APAR";
            var folder = DriveApp.getFoldersByName(folderName).hasNext() ? 
                         DriveApp.getFoldersByName(folderName).next() : 
                         DriveApp.createFolder(folderName);
            var base64Data = item.foto.split("base64,")[1];
            var blob = Utilities.newBlob(Utilities.base64Decode(base64Data), "image/jpeg", "EVIDEN_" + (item.kode || item.id || "APAR") + "_" + i + ".jpg");
            var file = folder.createFile(blob);
            file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
            photoUrl = file.getUrl();
          } catch (eImg) {
            if (photoUrl === "-") photoUrl = toSafeCell(item.photoUrl);
          }
        }

        rowsToAdd.push([
          toSafeCell(item.tanggal || new Date().toLocaleString("id-ID")),
          toSafeCell(item.kode || item.id),
          toSafeCell(item.gedung),
          toSafeCell(item.ruangan),
          toSafeCell(item.lantai),
          toSafeCell(item.merk),
          toSafeCell(item.kapasitas),
          toSafeCell(item.jenisGas),
          toSafeCell(item.kebersihanTabung || item.kebersihan),
          toSafeCell(item.indikatorTekanan || item.tekanan),
          toSafeCell(item.kunciPengaman || item.pin),
          toSafeCell(item.selangSemprot || item.selang),
          toSafeCell(item.nozzle),
          toSafeCell(item.tagLabel || item.tag),
          toSafeCell(item.status || "OK"),
          toSafeCell(item.keterangan),
          photoUrl,
          toSafeCell(item.pemeriksa || "Petugas")
        ]);
      }

      if (rowsToAdd.length > 0) {
        sheet.getRange(2, 1, rowsToAdd.length, 18).setValues(rowsToAdd);
      }

      return ContentService.createTextOutput(JSON.stringify({
        success: true,
        message: "Berhasil sinkronisasi massal " + rowsToAdd.length + " data inspeksi ke Google Sheets!",
        totalSynced: rowsToAdd.length
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // B. FITUR HAPUS TUGAS (action === "delete")
    if (data.action === "delete") {
      var targetKode = String(data.kode || data.id || "").trim().toLowerCase();
      if (targetKode) {
        var rows = sheet.getDataRange().getValues();
        var deletedCount = 0;
        for (var i = rows.length - 1; i >= 1; i--) {
          var rowKode = String(rows[i][1] || "").trim().toLowerCase();
          if (rowKode === targetKode) {
            sheet.deleteRow(i + 1);
            deletedCount++;
          }
        }
        return ContentService.createTextOutput(JSON.stringify({ 
          success: true, 
          message: "Data APAR Kode " + targetKode + " berhasil dihapus dari Google Sheets!", 
          deletedCount: deletedCount 
        })).setMimeType(ContentService.MimeType.JSON);
      }
    }

    // C. SIMPAN / UPDATE TUNGGAL (UPSERT REAL-TIME)
    var photoUrl = "-";
    if (data.foto && typeof data.foto === "string" && data.foto.indexOf("base64,") !== -1) {
      try {
        var folderName = "Foto_Eviden_APAR";
        var folder = DriveApp.getFoldersByName(folderName).hasNext() ? 
                     DriveApp.getFoldersByName(folderName).next() : 
                     DriveApp.createFolder(folderName);
        var base64Data = data.foto.split("base64,")[1];
        var blob = Utilities.newBlob(Utilities.base64Decode(base64Data), "image/jpeg", "EVIDEN_" + (data.kode || data.id || "APAR") + "_" + Date.now() + ".jpg");
        var file = folder.createFile(blob);
        file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        photoUrl = file.getUrl();
      } catch (imgErr) { photoUrl = toSafeCell(data.photoUrl) !== "-" ? toSafeCell(data.photoUrl) : ("Err: " + imgErr.toString()); }
    } else if (data.photoUrl) {
      photoUrl = toSafeCell(data.photoUrl);
    }

    var targetKodeNew = String(data.kode || data.id || "").trim().toLowerCase();
    var existingRowIndex = -1;
    if (targetKodeNew) {
      var allRows = sheet.getDataRange().getValues();
      for (var r = 1; r < allRows.length; r++) {
        if (String(allRows[r][1] || "").trim().toLowerCase() === targetKodeNew) {
          existingRowIndex = r + 1;
          break;
        }
      }
    }

    var rowValues = [
      toSafeCell(data.tanggal || new Date().toLocaleString("id-ID")),
      toSafeCell(data.kode || data.id),
      toSafeCell(data.gedung),
      toSafeCell(data.ruangan),
      toSafeCell(data.lantai),
      toSafeCell(data.merk),
      toSafeCell(data.kapasitas),
      toSafeCell(data.jenisGas),
      toSafeCell(data.kebersihanTabung || data.kebersihan),
      toSafeCell(data.indikatorTekanan || data.tekanan),
      toSafeCell(data.kunciPengaman || data.pin),
      toSafeCell(data.selangSemprot || data.selang),
      toSafeCell(data.nozzle),
      toSafeCell(data.tagLabel || data.tag),
      toSafeCell(data.status || "OK"),
      toSafeCell(data.keterangan),
      photoUrl,
      toSafeCell(data.pemeriksa || "Petugas")
    ];

    if (existingRowIndex > 0) {
      sheet.getRange(existingRowIndex, 1, 1, 18).setValues([rowValues]);
    } else {
      sheet.appendRow(rowValues);
    }

    return ContentService.createTextOutput(JSON.stringify({ 
      success: true, 
      message: "Data APAR Kode " + (data.kode || data.id || "") + " berhasil dicatat ke Google Sheets & Drive!", 
      photoUrl: photoUrl 
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ 
      success: false, 
      error: err.toString() 
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
