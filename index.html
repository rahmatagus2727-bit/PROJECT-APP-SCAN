Code.gs 
/**
 * GOOGLE APPS SCRIPT - SISTEM PEMELIHARAAN APAR (KODE 1 - 128)
 * Format Kolom & Header 100% Persis Template Master Data FSS / APAR
 * 
 * CARA PASANG:
 * 1. Di Google Sheets: Klik Extensions > Apps Script
 * 2. Hapus semua isi, Paste kode ini, lalu klik Simpan (Ctrl+S)
 * 3. Klik Deploy > New deployment > Jenis: Web app
 *    - Execute as: Me (email Anda)
 *    - Who has access: Anyone (Wajib pilih "Anyone" agar bisa diakses aplikasi)
 * 4. Salin Web App URL (akhiran /exec) dan simpan di aplikasi.
 */

function doPost(e) {
  return handleRequest(e);
}

function doGet(e) {
  return handleRequest(e);
}

function handleRequest(e) {
  try {
    var rawData = "";
    if (e && e.postData && e.postData.contents) {
      rawData = e.postData.contents;
    } else if (e && e.parameter && e.parameter.data) {
      rawData = e.parameter.data;
    } else if (e && e.parameter && e.parameter.payload) {
      rawData = e.parameter.payload;
    }
    
    // Ping test check
    if (!rawData && (!e || !e.parameter || (!e.parameter.kode && !e.parameter.id && !e.parameter.data))) {
      return ContentService.createTextOutput(JSON.stringify({
        success: true,
        status: "active",
        message: "Webhook Google Apps Script APAR siap menerima data."
      })).setMimeType(ContentService.MimeType.JSON);
    }
    
    var payload = {};
    if (rawData) {
      try {
        payload = JSON.parse(rawData);
      } catch(err) {
        payload = { raw: rawData };
      }
    } else if (e && e.parameter) {
      payload = e.parameter;
    }
    
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) {
      return ContentService.createTextOutput(JSON.stringify({
        success: false,
        error: "Spreadsheet tidak aktif. Buka script dari menu Extensions > Apps Script."
      })).setMimeType(ContentService.MimeType.JSON);
    }

    var sheet = ss.getSheetByName("DATA FSS & APAR");
    if (!sheet) {
      sheet = ss.getSheetByName("Inspeksi APAR") || ss.getActiveSheet();
      sheet.setName("DATA FSS & APAR");
    }
    
    // Inisialisasi Header Standar jika sheet baru/kosong
    initHeaderIfEmpty(sheet);
    
    // Siapkan folder foto Google Drive
    var folderName = "Foto_Inspeksi_APAR";
    var folders = DriveApp.getFoldersByName(folderName);
    var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);
    
    var items = Array.isArray(payload.items) ? payload.items : [payload];
    var processedCount = 0;
    
    items.forEach(function(item) {
      if (!item || typeof item !== 'object') return;
      var kodeStr = String(item.kode || item.id || "").trim();
      if (!kodeStr) return;
      
      var photoUrl = "";
      
      // Upload Foto Base64 ke Google Drive
      if (item.foto && typeof item.foto === 'string' && item.foto.indexOf("data:image") === 0) {
        try {
          var parts = item.foto.split(",");
          var contentType = parts[0].split(":")[1].split(";")[0];
          var base64Data = parts[1];
          var decoded = Utilities.base64Decode(base64Data);
          var fileName = "APAR_" + kodeStr + "_" + Utilities.formatDate(new Date(), "Asia/Jakarta", "yyyyMMdd_HHmmss") + ".jpg";
          var blob = Utilities.newBlob(decoded, contentType, fileName);
          var file = folder.createFile(blob);
          file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
          photoUrl = file.getUrl();
        } catch (imgErr) {
          photoUrl = "Error upload foto: " + imgErr.toString();
        }
      } else if (item.photoDriveUrl) {
        photoUrl = item.photoDriveUrl;
      } else if (item.foto && typeof item.foto === 'string' && item.foto.indexOf("http") === 0) {
        photoUrl = item.foto;
      }
      
      var kebersihan = item.kebersihanTabung === "good" ? "Bersih" : (item.kebersihanTabung === "bad" ? "Kotor" : (item.kebersihanTabung || "-"));
      var tekanan = item.indikatorTekanan === "good" ? "Hijau" : (item.indikatorTekanan === "bad" ? "Merah" : (item.indikatorTekanan || "-"));
      var kunci = item.kunciPengaman === "good" ? "Ada" : (item.kunciPengaman === "bad" ? "Tidak Ada" : (item.kunciPengaman || "-"));
      var selang = item.selangSemprot === "good" ? "Tidak Pecah" : (item.selangSemprot === "bad" ? "Pecah" : (item.selangSemprot || "-"));
      var nozzle = item.nozzle === "good" ? "Ada" : (item.nozzle === "bad" ? "Tidak Ada" : (item.nozzle || "-"));
      var tag = item.tagLabel === "good" ? "Ada" : (item.tagLabel === "bad" ? "Tidak Ada" : (item.tagLabel || "-"));
      var tgl = item.tanggal ? Utilities.formatDate(new Date(item.tanggal), "Asia/Jakarta", "dd/MM/yyyy HH:mm") : Utilities.formatDate(new Date(), "Asia/Jakarta", "dd/MM/yyyy HH:mm");
      var fotoFormula = photoUrl ? ('=IMAGE("' + photoUrl + '")') : (item.foto ? "📷 Ada Foto" : "-");

      var rowValues = [
        kodeStr,
        item.ruangan || "-",
        item.gedung || "-",
        item.lantai || "-",
        item.merk || "-",
        item.kapasitas || "-",
        item.jenisGas || "-",
        kebersihan,
        tekanan,
        kunci,
        selang,
        nozzle,
        tag,
        item.keterangan || "-",
        fotoFormula,
        photoUrl || (item.foto ? "Tersimpan di Aplikasi" : "-"),
        tgl,
        item.pemeriksa || "Petugas",
        item.status || "✅ Selesai"
      ];
      
      // Update baris yang sudah ada jika kode ditemukan di Kolom A, atau tambahkan di bawah
      var targetRow = findRowByKode(sheet, kodeStr);
      if (targetRow > 2) {
        sheet.getRange(targetRow, 1, 1, rowValues.length).setValues([rowValues]);
      } else {
        sheet.appendRow(rowValues);
      }
      processedCount++;
    });
    
    return ContentService.createTextOutput(JSON.stringify({
      success: true,
      message: "Berhasil memproses " + processedCount + " data APAR ke Google Sheets.",
      spreadsheetUrl: ss.getUrl(),
      folderUrl: folder.getUrl(),
      processedCount: processedCount
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      success: false,
      error: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function findRowByKode(sheet, kode) {
  var lastRow = sheet.getLastRow();
  if (lastRow <= 2) return -1;
  var colA = sheet.getRange(3, 1, lastRow - 2, 1).getValues();
  for (var i = 0; i < colA.length; i++) {
    if (String(colA[i][0]).trim() === String(kode).trim()) {
      return i + 3;
    }
  }
  return -1;
}

function initHeaderIfEmpty(sheet) {
  if (sheet.getLastRow() > 0) return;
  
  // Row 1: Merged Headers
  var row1 = [
    "APAR", "", "", "", "", "", "",
    "KONDISI", "", "", "", "", "",
    "KETERANGAN",
    "FOTO BUKTI (EVIDEN)",
    "LINK FOTO DRIVE",
    "TANGGAL PEMERIKSAAN",
    "PETUGAS PEMERIKSA",
    "STATUS"
  ];
  
  // Row 2: Sub Headers
  var row2 = [
    "KODE",
    "RUANGAN",
    "GEDUNG",
    "LANTAI",
    "MERK",
    "KAPASITAS",
    "JENIS GAS",
    "Kebersihan Tabung APAR",
    "Pemeriksaan Indikator Tekanan",
    "Pemeriksaan Kunci Pengaman",
    "Pemeriksaan Selang Semprot",
    "Pemeriksaan Nozzle",
    "TAG Label Pemeliharaan",
    "KETERANGAN",
    "FOTO BUKTI (EVIDEN)",
    "LINK FOTO DRIVE",
    "TANGGAL PEMERIKSAAN",
    "PETUGAS PEMERIKSA",
    "STATUS"
  ];
  
  sheet.appendRow(row1);
  sheet.appendRow(row2);
  
  // Merge Headers
  sheet.getRange("A1:G1").merge().setHorizontalAlignment("center").setVerticalAlignment("middle");
  sheet.getRange("H1:M1").merge().setHorizontalAlignment("center").setVerticalAlignment("middle");
  sheet.getRange("N1:N2").merge().setHorizontalAlignment("center").setVerticalAlignment("middle");
  sheet.getRange("O1:O2").merge().setHorizontalAlignment("center").setVerticalAlignment("middle");
  sheet.getRange("P1:P2").merge().setHorizontalAlignment("center").setVerticalAlignment("middle");
  sheet.getRange("Q1:Q2").merge().setHorizontalAlignment("center").setVerticalAlignment("middle");
  sheet.getRange("R1:R2").merge().setHorizontalAlignment("center").setVerticalAlignment("middle");
  sheet.getRange("S1:S2").merge().setHorizontalAlignment("center").setVerticalAlignment("middle");
  
  // Row 1 Colors
  sheet.getRange("A1:G1").setBackground("#1e293b").setFontColor("#ffffff").setFontWeight("bold");
  sheet.getRange("H1:M1").setBackground("#c00000").setFontColor("#ffffff").setFontWeight("bold");
  sheet.getRange("N1:N2").setBackground("#c00000").setFontColor("#ffffff").setFontWeight("bold");
  sheet.getRange("O1:S2").setBackground("#1e293b").setFontColor("#ffffff").setFontWeight("bold");
  
  // Row 2 Colors
  sheet.getRange("A2:G2").setBackground("#1e293b").setFontColor("#ffffff").setFontWeight("bold").setHorizontalAlignment("center");
  sheet.getRange("H2:M2").setBackground("#c00000").setFontColor("#ffffff").setFontWeight("bold").setHorizontalAlignment("center");
  
  sheet.setFrozenRows(2);
}
